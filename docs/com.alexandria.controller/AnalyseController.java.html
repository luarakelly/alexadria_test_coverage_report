package com.alexandria.controller;

import java.sql.SQLException;
import java.util.Comparator;
import java.util.List;

import com.alexandria.dao.TermAnalysisDAO;
import com.alexandria.dao.TextAnalysisDAO;
import com.alexandria.model.TermAnalysis;
import com.alexandria.model.Text;
import com.alexandria.model.TextAnalysis;
import com.alexandria.model.User;
import com.alexandria.service.analysis.*;
import com.alexandria.utils.JsonMapper;

/**
 * Controller for text analysis and search. Pure business logic — no View
 * imports at all. Returns service-level result types (TextAnalysisResult,
 * TermAnalysisResult, SearchMatch); translating those into view-local types
 * is done by whoever wires this controller (MainController).
 */
public class AnalyseController {

    private final SearchServiceINT searchService;
    private final TermAnalysisServiceINT termAnalysisService;
    private final TextAnalysisServiceINT textAnalysisService;
    private final TextAnalysisDAO textAnalysisDAO;
    private final TermAnalysisDAO termAnalysisDAO;
    private final UserSessionController session = UserSessionController.getInstance();

    private Text currentText;

    public AnalyseController(
            SearchServiceINT searchService,
            TermAnalysisServiceINT termAnalysisService,
            TextAnalysisServiceINT textAnalysisService,
            TextAnalysisDAO textAnalysisDAO,
            TermAnalysisDAO termAnalysisDAO) {
        this.searchService = searchService;
        this.termAnalysisService = termAnalysisService;
        this.textAnalysisService = textAnalysisService;
        this.textAnalysisDAO = textAnalysisDAO;
        this.termAnalysisDAO = termAnalysisDAO;
    }

    /** Sets the active text and loads (or reuses cached) whole-text analysis. */
    public TextAnalysisOutcome openText(Text text) {
        this.currentText = text;
        try {
            TextAnalysis existing = findLatestTextAnalysis(text.getId());

            TextAnalysisResult result = existing != null
                    ? JsonMapper.fromJson(existing.getAnalysisData(), TextAnalysisResult.class)
                    : computeAndPersistTextAnalysis();

            return TextAnalysisOutcome.ok(result);
        } catch (SQLException e) {
            return TextAnalysisOutcome.error("Could not load text analysis: " + e.getMessage());
        }
    }

    /** Runs a search against the currently open text and loads (or reuses cached) term analysis. */
    public SearchOutcome search(String term, boolean caseSensitive, boolean fuzzy, boolean wholeWordsOnly) {
        if (currentText == null) {
            return SearchOutcome.error("No text is currently open.");
        }

        try {
            SearchSettings settings = new SearchSettings(caseSensitive, fuzzy, wholeWordsOnly);
            List<SearchMatch> matches = searchService.search(currentText.getContent(), term, settings);

            TermAnalysis existing = findLatestTermAnalysis(currentText.getId(), term);
            TermAnalysisResult termResult = existing != null
                    ? JsonMapper.fromJson(existing.getAnalysisData(), TermAnalysisResult.class)
                    : computeAndPersistTermAnalysis(term);

            return SearchOutcome.ok(matches, termResult);
        } catch (SQLException e) {
            return SearchOutcome.error("Could not complete search: " + e.getMessage());
        }
    }

    /* Business logic */

    private TextAnalysisResult computeAndPersistTextAnalysis() throws SQLException {
        TextAnalysisResult result = textAnalysisService.analyzeText(currentText.getContent());

        Integer userId = currentUserId();
        if (userId != null) {
            textAnalysisDAO.create(new TextAnalysis(userId, currentText.getId(), JsonMapper.toJson(result)));
        }
        return result;
    }

    private TermAnalysisResult computeAndPersistTermAnalysis(String term) throws SQLException {
        TermAnalysisResult result = termAnalysisService.analyzeTerm(currentText.getContent(), term);

        Integer userId = currentUserId();
        if (userId != null) {
            termAnalysisDAO.create(new TermAnalysis(userId, currentText.getId(), term, JsonMapper.toJson(result)));
        }
        return result;
    }

    private TextAnalysis findLatestTextAnalysis(int textId) throws SQLException {
        return textAnalysisDAO.findAllByTextId(textId).stream()
                .max(Comparator.comparing(TextAnalysis::getCreatedAt))
                .orElse(null);
    }

    private TermAnalysis findLatestTermAnalysis(int textId, String term) throws SQLException {
        return termAnalysisDAO.findAllByTextId(textId).stream()
                .filter(row -> row.getTerm().equalsIgnoreCase(term))
                .max(Comparator.comparing(TermAnalysis::getCreatedAt))
                .orElse(null);
    }

    private Integer currentUserId() {
        User user = session.getCurrentUser();
        return user != null ? user.getId() : null;
    }

    /* Outcomes — same Result-style pattern as ProjectController */

    public record TextAnalysisOutcome(boolean success, String message, TextAnalysisResult result) {
        static TextAnalysisOutcome ok(TextAnalysisResult result) { return new TextAnalysisOutcome(true, null, result); }
        static TextAnalysisOutcome error(String message) { return new TextAnalysisOutcome(false, message, null); }
    }

    public record SearchOutcome(boolean success, String message, List<SearchMatch> matches, TermAnalysisResult termAnalysis) {
        static SearchOutcome ok(List<SearchMatch> matches, TermAnalysisResult termAnalysis) {
            return new SearchOutcome(true, null, matches, termAnalysis);
        }
        static SearchOutcome error(String message) { return new SearchOutcome(false, message, null, null); }
    }
}
